reglas z-rendi viejas v1

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Usuarios
    match /usuarios/{userId} {
      allow read: if request.auth != null && (
        request.auth.uid == userId || request.auth.token.role == "admin"
      );
      allow write: if request.auth != null && (
        request.auth.uid == userId || request.auth.token.role == "admin"
      );
    }

   // Fichas
match /fichas/{docId} {
  // El cliente puede leer su propia ficha (docId == uid) o el admin puede leer cualquiera
  allow read: if request.auth != null && (
    request.auth.uid == docId || 
    (resource != null && request.auth.uid == resource.data.clienteId) || 
    request.auth.token.role == "admin"
  );
  
  // El cliente puede crear/actualizar su propia ficha si su UID es el ID del documento
  // O el administrador puede crear/actualizar/borrar cualquier ficha
  allow create, update: if request.auth != null && (
    request.auth.uid == docId || 
    request.auth.token.role == "admin"
  );
  
  // Solo el administrador puede borrar fichas
  allow delete: if request.auth != null && request.auth.token.role == "admin";
}

    // Cuotas
    match /cuotas/{docId} {
      allow read: if request.auth != null && (
        request.auth.uid == resource.data.clienteId || request.auth.token.role == "admin"
      );
      allow write: if request.auth != null && request.auth.token.role == "admin";
    }
    
    // Asistencias
match /asistencias/{docId} {
  allow read: if request.auth != null && (
    request.auth.uid == resource.data.clienteId ||
    request.auth.token.role == "admin"
  );

  allow create, update, delete: if request.auth != null &&
    request.auth.token.role == "admin";
}

    // Sesiones
match /sesiones/{docId} {

  allow read: if request.auth != null && (
    request.auth.uid == resource.data.clienteId ||
    request.auth.token.role == "admin"
  );

  allow update: if request.auth != null &&
    request.auth.token.role == "admin";

  allow create: if request.auth != null && (
    request.auth.token.role == "admin" ||
    (
      request.auth.uid == request.resource.data.clienteId &&
      exists(
        /databases/$(database)/documents/sesionesPendientes/$(docId)
      )
    )
  );

  allow delete: if request.auth != null &&
    request.auth.token.role == "admin";
}

    // Clientes
    match /clientes/{docId} {
      allow read: if request.auth != null && request.auth.token.role == "admin";
      allow write: if request.auth != null && request.auth.token.role == "admin";
    }

    // Guías de ejercicios
    match /guias/{docId} {
      allow read: if request.auth != null;
      allow create, update, delete: if request.auth != null && request.auth.token.role == "admin";
    }
    
    // Biblioteca de libros de entrenamiento
match /trainingBooks/{bookId} {

  // Cada entrenador puede ver solamente sus propios libros
  allow read: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;


  // Puede crear libros asignados a su propio UID
  allow create: if request.auth != null &&
    request.resource.data.preparadorId == request.auth.uid;


  // Puede modificar o eliminar solamente sus libros
  allow update, delete: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;
}

	
		// BARRERAS
match /barrerasProgreso/{docId} {
  allow read: if request.auth != null && (
    resource.data.clienteId == request.auth.uid ||
    resource.data.preparadorId == request.auth.uid
  );
  allow create: if request.auth != null &&
    request.resource.data.preparadorId == request.auth.uid;
  allow update: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid &&
    request.resource.data.preparadorId == resource.data.preparadorId;
  allow delete: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;
}


    //TEST FISICOS
    match /physicalTests/{docId} {

  allow read: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;

  allow create: if request.auth != null &&
    request.resource.data.preparadorId == request.auth.uid;

  allow update, delete: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;
}

// BIBLIOTECA DE EJERCICIOS
match /exerciseLibrary/{exerciseId} {

  // Cada preparador puede ver solamente sus propios ejercicios
  allow read: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;

  // Puede crear ejercicios asignados a su propio UID
  allow create: if request.auth != null &&
    request.resource.data.preparadorId == request.auth.uid;

  // Puede modificar o eliminar solamente sus propios ejercicios
  allow update, delete: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;
}



// SESIONES PENDIENTES
match /sesionesPendientes/{docId} {

  allow create: if request.auth != null &&
    request.resource.data.preparadorId == request.auth.uid;

  allow read: if request.auth != null && (
    request.auth.uid == resource.data.clienteId ||
    request.auth.uid == resource.data.preparadorId
  );

  allow update: if request.auth != null && (
    request.auth.uid == resource.data.clienteId ||
    request.auth.uid == resource.data.preparadorId
  );

  allow delete: if request.auth != null && (
    request.auth.uid == resource.data.preparadorId ||
    request.auth.uid == resource.data.clienteId
  );
}

// INVITACIONES
match /invitaciones/{invitacionId} {

  // El preparador puede crear sus propias invitaciones
  allow create: if request.auth != null &&
    request.resource.data.preparadorId == request.auth.uid;

  // El preparador puede leer y eliminar sus propias invitaciones
  allow read, delete: if request.auth != null &&
    resource.data.preparadorId == request.auth.uid;

  // Una persona no autenticada puede consultar
  // únicamente la invitación cuyo token coincide con el ID.
  allow read: if request.auth == null &&
    resource.data.token == invitacionId;

  // El cliente recién registrado puede marcar
  // su invitación como usada.
  // Solo puede modificar estado, clienteId y usedAt.
  allow update: if request.auth != null &&
    resource.data.estado == "pendiente" &&
    request.resource.data.estado == "usada" &&
    request.resource.data.clienteId == request.auth.uid &&
    request.resource.data.diff(resource.data).affectedKeys()
      .hasOnly(["estado", "clienteId", "usedAt"]);
}


    // Planillas y seguimiento (solo admin)
    match /planillas/{planillaId} {
      // Solo admin puede crear, leer, actualizar y borrar planillas
      allow create, read, update, delete: if request.auth != null && request.auth.token.role == "admin";

      // Subcolecciones (atletas, registros, etc.) accesibles solo por admin
      match /{subCollection}/{docId} {
        allow read, create, update, delete: if request.auth != null && request.auth.token.role == "admin";
      }
    }
    // EQUIPOS / PLANTELES
    match /equipos/{equipoId} {

      // Leer: el preparador ve solo sus equipos (o el admin)
      allow read: if request.auth != null && (
        resource.data.preparadorId == request.auth.uid ||
        request.auth.token.role == "admin"
      );

      // Crear: cualquier usuario autenticado puede crear un equipo si se asigna como preparador
      allow create: if request.auth != null && (
        request.resource.data.preparadorId == request.auth.uid ||
        request.auth.token.role == "admin"
      );

      // Actualizar y eliminar: solo el creador (preparador) o el admin
      allow update, delete: if request.auth != null && (
        resource.data.preparadorId == request.auth.uid ||
        request.auth.token.role == "admin"
      );

      // Subcolecciones de equipos (atletas del plantel, bloques de sesión, etc.)
      match /{subColeccion}/{docId} {
        allow read, write: if request.auth != null;
      }
    }

    // SESIONES DE PLANTEL
    match /sesionesPlantel/{sesionId} {
      allow read: if request.auth != null && (
        resource.data.preparadorId == request.auth.uid ||
        request.auth.token.role == "admin"
      );

      allow create: if request.auth != null && (
        request.resource.data.preparadorId == request.auth.uid ||
        request.auth.token.role == "admin"
      );

      allow update, delete: if request.auth != null && (
        resource.data.preparadorId == request.auth.uid ||
        request.auth.token.role == "admin"
      );
    }

    // REGISTROS DE EJERCICIO
match /registrosEjercicio/{docId} {
  allow read: if request.auth != null && (
    request.auth.uid == resource.data.clienteId ||
    request.auth.uid == resource.data.preparadorId ||
    request.auth.token.role == "admin"
  );
  allow create: if request.auth != null && (
    request.auth.uid == request.resource.data.clienteId ||
    request.auth.uid == request.resource.data.preparadorId
  );
  allow update, delete: if request.auth != null &&
    request.auth.uid == resource.data.preparadorId;
}

// CHECK-INS
match /checkins/{docId} {
  allow read: if request.auth != null && (
    docId.matches(request.auth.uid + '_.*') ||
    (resource != null && resource.data.preparadorId == request.auth.uid)
  );
  allow create, update: if request.auth != null &&
    docId.matches(request.auth.uid + '_.*') &&
    request.resource.data.clienteId == request.auth.uid;
  allow delete: if false;
}

        // RELEVAMIENTOS
    match /relevamientos/{relevamientoId} {

      // Cada preparador puede leer solamente sus propios relevamientos
      allow read: if request.auth != null &&
        resource.data.preparadorId == request.auth.uid;

      // Puede crear un relevamiento asignado a su propio UID
      allow create: if request.auth != null &&
        request.resource.data.preparadorId == request.auth.uid;

      // Puede modificar o eliminar solamente sus propios relevamientos
      allow update, delete: if request.auth != null &&
        resource.data.preparadorId == request.auth.uid;
    }
  }
}
